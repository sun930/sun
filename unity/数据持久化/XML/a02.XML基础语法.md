## 基本语法
```xml
<!--XML的基本语法 就是 <元素标签> </元素标签>配对出现-->
<PlayerInfo>
	<name>野生狗奶过期</name>
	<age>22</age>
	<sex>true</sex>
	<ItemList>
		<Item>
			<id>1</id>
			<num>10</num>
		</Item>
		<Item>
			<id>2</id>
			<num>10</num>
		</Item>
	</ItemList>
</PlayerInfo>
```
